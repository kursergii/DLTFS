cmake_minimum_required(VERSION 3.5)

project(DLTFS VERSION 0.1 LANGUAGES CXX)

set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Qt6 REQUIRED COMPONENTS Widgets PrintSupport)

set(PROJECT_SOURCES
    main.cpp
    mw.cpp
    mw.h
    mw.ui
    gpib/gpibdevice.cpp
    gpib/gpibdevice.h
    gpib/hp4291analyzer.cpp
    gpib/hp4291analyzer.h
    gpib/keithley236smu.cpp
    gpib/keithley236smu.h
    qcustomplot/qcustomplot.cpp
    qcustomplot/qcustomplot.h
    picoscope/picoscope.cpp
    picoscope/picoscope.h
)

add_executable(DLTFS ${PROJECT_SOURCES}
    measurements.h measurements.cpp)

# Add include directories so autogenerated UI can find qcustomplot.h and gpib headers
target_include_directories(DLTFS PRIVATE
    ${CMAKE_SOURCE_DIR}
    ${CMAKE_SOURCE_DIR}/picoscope/include/libps4000a
)

target_link_directories(DLTFS PRIVATE ${CMAKE_SOURCE_DIR}/picoscope/lib)

target_link_libraries(DLTFS PRIVATE
    Qt6::Widgets
    Qt6::PrintSupport
    gpib
    ps4000a
)
