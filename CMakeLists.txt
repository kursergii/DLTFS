cmake_minimum_required(VERSION 3.5)

project(DLTFS VERSION 0.1 LANGUAGES CXX)

set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Qt6 REQUIRED COMPONENTS Widgets SerialPort PrintSupport)

set(PROJECT_SOURCES
    main.cpp
    mw.cpp
    mw.h
    mw.ui
    gpib/gpibdevice.cpp
    gpib/gpibdevice.h
    gpib/hp4291analyzer.cpp
    gpib/hp4291analyzer.h
    gpib/hp8114apulser.cpp
    gpib/hp8114apulser.h
    qcustomplot.cpp
    qcustomplot.h
)

add_executable(DLTFS ${PROJECT_SOURCES}
    measurements.h measurements.cpp)

# Add include directories so autogenerated UI can find qcustomplot.h and gpib headers
target_include_directories(DLTFS PRIVATE ${CMAKE_SOURCE_DIR})

target_link_libraries(DLTFS PRIVATE
    Qt6::Widgets
    Qt6::SerialPort
    Qt6::PrintSupport
    gpib
)
